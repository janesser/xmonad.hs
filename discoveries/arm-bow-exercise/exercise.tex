% ===== PAGE 2: the steps =====
\begin{frame}
\centering{\bfseries\large How to do it}\vspace{0.4em}

{\normalsize
\begin{enumerate}[leftmargin=2.2em,itemsep=0.45em,topsep=0.2em,label=\bfseries\arabic*.]
  \item Sit up straight in a comfortable chair.
  \item Let your arm hang relaxed by your side.
  \item Slowly bend your elbow, raising your forearm toward your shoulder.
  \item Hold for 3 to 5 seconds.
  \item Slowly lower your arm back down.
  \item Repeat as your therapist directs.
\end{enumerate}
}

\vspace{0.5em}
{\normalsize\emph{Tip:\quad Move gently.\quad Do not push into pain.\quad Breathe normally.}}
